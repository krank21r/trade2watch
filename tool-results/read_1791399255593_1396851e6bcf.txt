    64→/** One-chip summary per asset for the ZoneWatch strip. */
    65→function summarize(s: BoardSetup): { cls: string; txt: string } {
    66→  const l = s.long
    67→  const sh = s.short
    68→  if (sh?.state === 'LIVE') return { cls: chipCls.liveShort, txt: 'SHORT CONFIRMED' }
    69→  if (l?.state === 'LIVE') return { cls: chipCls.liveLong, txt: 'LONG CONFIRMED' }
    70→  const waits: string[] = []
    71→  if (l?.state === 'WAITING' && l.distPct !== null) waits.push(`${l.distPct.toFixed(1)}% to LONG`)
    72→  if (sh?.state === 'WAITING' && sh.distPct !== null) waits.push(`${sh.distPct.toFixed(1)}% to SHORT`)
    73→  const voids: string[] = []
    74→  if (l?.state === 'VOID') voids.push('LONG void')
    75→  if (sh?.state === 'VOID') voids.push('SHORT void')
    76→  if (voids.length === 2) return { cls: chipCls.void, txt: 'SETUP VOID' }
    77→  if (waits.length === 0 && voids.length === 1) return { cls: chipCls.void, txt: `${voids[0].toUpperCase()} — WAIT` }
    78→  if (voids.length === 1) return { cls: chipCls.waiting, txt: `${waits[0]} · ${voids[0]}` }
    79→  return { cls: chipCls.waiting, txt: waits.length ? `WAITING · ${waits.join(' · ')}` : 'WAITING' }
    80→}

    84→function pctFrom(mid: number, v: number): string {
    85→  const p = ((v - mid) / mid) * 100
    86→  return `${p >= 0 ? '+' : ''}${p.toFixed(1)}%`
    87→}
    88→
    89→function rMult(v: number, mid: number, risk: number): string {
    90→  return risk > 0 ? `${(Math.abs(v - mid) / risk).toFixed(1)}R` : '—'
    91→}

   422→export function SetupBoard() {
   423→  const [symbols, setSymbols] = useState<string[]>(['BTC', 'ETH', 'SOL'])
   424→  const [data, setData] = useState<SetupsPayload | null>(null)
   425→  const [loading, setLoading] = useState(true)
   426→  const [error, setError] = useState<string | null>(null)
   427→  const [selected, setSelected] = useState('BTC')
   428→  const [input, setInput] = useState('')
   429→  const [dismissed, setDismissed] = useState<Set<string>>(new Set())
   430→  const [confirms, setConfirms] = useState<Record<string, ConfirmInfo>>({})
   431→  const [acct, setAcct] = useState('')
   432→  const [riskPct, setRiskPct] = useState('1')
   433→  const mounted = useRef(true)
   434→  const confirmsRef = useRef<Record<string, ConfirmInfo>>({})
   435→  const { toast } = useToast()
   436→
   437→  useEffect(() => {
   438→    mounted.current = true
   439→    setAcct(lsGet('tw_acct', ''))
   440→    setRiskPct(lsGet('tw_risk', '1'))
   441→    const c = loadConfirms()
   442→    confirmsRef.current = c
   443→    setConfirms(c)
   444→    return () => {
   445→      mounted.current = false
   446→    }
   447→  }, [])
   448→
   449→  const load = useCallback(async () => {
   450→    try {
   451→      const res = await fetch(`/api/setups?symbols=${symbols.join(',')}`, { cache: 'no-store' })
   452→      if (!res.ok) throw new Error(`API ${res.status}`)
   453→      const d = (await res.json()) as SetupsPayload
   454→      if (!mounted.current) return
   455→      setData(d)
   456→      setError(null)
   457→    } catch (e) {
   458→      if (mounted.current) setError(String(e).slice(0, 120))
   459→    } finally {
   460→      if (mounted.current) setLoading(false)
   461→    }
   462→  }, [symbols])
   463→
   464→  useEffect(() => {
   465→    setLoading(true)
   466→    load()
   467→    const t = setInterval(load, POLL_MS)
   468→    return () => clearInterval(t)
   469→  }, [load])
   470→
   471→  // confirmation tracker — when price reaches an entry zone, record it once,
   472→  // fire a toast, and re-arm automatically if the zone is left/voided
   473→  useEffect(() => {
   474→    if (!data) return
   475→    const prev = confirmsRef.current
   476→    const next: Record<string, ConfirmInfo> = {}
   477→    const fired: Array<{ key: string; symbol: string; side: 'long' | 'short'; info: ConfirmInfo; s: SideSetup; setup: BoardSetup }> = []
   478→    for (const setup of data.setups) {
   479→      for (const kind of ['long', 'short'] as const) {
   480→        const s = setup[kind]
   481→        if (!s) continue
   482→        const key = `${setup.symbol}:${kind}`
   483→        if (s.state === 'LIVE') {
   484→          const prior = prev[key]
