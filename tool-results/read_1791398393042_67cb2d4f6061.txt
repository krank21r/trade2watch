    65→    65→function summarize(s: BoardSetup): { cls: string; txt: string } {
    66→    66→  const l = s.long
    67→    67→  const sh = s.short
    68→    68→  if (sh?.state === 'LIVE') return { cls: chipCls.liveShort, txt: 'SHORT CONFIRMED' }
    69→    69→  if (l?.state === 'LIVE') return { cls: chipCls.liveLong, txt: 'LONG CONFIRMED' }
    70→    70→  const waits: string[] = []
    71→    71→  if (l?.state === 'WAITING' && l.distPct !== null) waits.push(`${l.distPct.toFixed(1)}% to LONG`)
    72→    72→  if (sh?.state === 'WAITING' && sh.distPct !== null) waits.push(`${sh.distPct.toFixed(1)}% to SHORT`)
    73→    73→  const voids: string[] = []
    74→    74→  if (l?.state === 'VOID') voids.push('LONG void')
    75→    75→  if (sh?.state === 'VOID') voids.push('SHORT void')
    76→    76→  if (voids.length === 2) return { cls: chipCls.void, txt: 'SETUP VOID' }
    77→    77→  if (waits.length === 0 && voids.length === 1) return { cls: chipCls.void, txt: `${voids[0].toUpperCase()} — WAIT` }
    78→    78→  if (voids.length === 1) return { cls: chipCls.waiting, txt: `${waits[0]} · ${voids[0]}` }
    79→    79→  return { cls: chipCls.waiting, txt: waits.length ? `WAITING · ${waits.join(' · ')}` : 'WAITING' }
    80→    80→}
