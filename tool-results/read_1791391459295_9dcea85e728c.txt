   680→  function render() {

   684→      var apiKey = key === 'btc' ? 'bitcoin' : key === 'eth' ? 'ethereum' : key;

   695→
   696→      // gauge

   698→      var pos = Math.min(1, Math.max(0, (p - g.min) / (g.max - g.min)));
   699→      $(key + 'Marker').style.left = (pos * 100) + '%';
   700→      $(key + 'GaugePos').textContent = 'PRICE ' + money(p);

   710→        $('zw' + key.toUpperCase() + 'Dist').textContent = distToNext(a.zones, p);
   711→      }
   712→
   713→      // state lines (re-evaluate with current price)
   714→      var ss = $(key + 'ShortState'), ls = $(key + 'LongState');
   715→      if (ss && ls) {
   716→        var zs2 = zoneState(a.zones, p);
   717→        if (zs2.cls.indexOf('hot') === 0 && zs2.txt.indexOf('SHORT') === 0) {
   718→          ss.className = 'state-line hot';
   719→          ss.textContent = '🔴 SHORT ZONE ACTIVE — price ≥ ' + money(a.zones.shortEnter) + '. Limit ' + zoneRange(a, 'short') + ', stop ' + money(a.short.stop) + ', T1 ' + money(a.short.t1) + ', T2 ' + money(a.short.t2) + '.';
   720→          ls.className = 'state-line wait'; ls.textContent = '⏳ Short zone live — long not in play';
   721→        } else if (zs2.cls.indexOf('hot') === 0) {

   761→  function beep() {
   762→    if (!audioCtx) return;
   763→    try {
   764→      var t0 = audioCtx.currentTime;
   765→      [0, .2, .4].forEach(function (d) {
   766→        var o = audioCtx.createOscillator();
   767→        var g = audioCtx.createGain();
   768→        o.type = 'square'; o.frequency.value = 880;
   769→        g.gain.setValueAtTime(.05, t0 + d);
   770→        g.gain.exponentialRampToValueAtTime(.0001, t0 + d + .14);
   771→        o.connect(g); g.connect(audioCtx.destination);
   772→        o.start(t0 + d); o.stop(t0 + d + .15);
   773→      });
   774→    } catch (e) {}
   775→  }

   843→  function poll() {
   844→    var ids = Object.keys(SETUP.assets).map(function (k) { return k === 'btc' ? 'bitcoin' : k === 'eth' ? 'ethereum' : k; }).join(',');
   845→    var ctrl = new AbortController();
   846→    var t = setTimeout(function () { ctrl.abort(); }, 12000);
   847→    fetch('https://api.coingecko.com/api/v3/simple/price?ids=' + ids + '&vs_currencies=usd&include_24hr_change=true', { signal: ctrl.signal })
   848→      .then(function (r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
   849→      .then(function (d) {
   850→        clearTimeout(t);
   851→        readSetup(); // pick up pushed setups without refresh
   852→        if (SETUP.updated !== lastUpdated) build();
   853→        lastFetchAt = Date.now();
   854→        feedOk = true;
   855→        Object.keys(SETUP.assets).forEach(function (key) {
   856→          var apiKey = key === 'btc' ? 'bitcoin' : key === 'eth' ? 'ethereum' : key;
   857→          if (!d[apiKey]) return;
   858→          state[apiKey] = d[apiKey].usd;
   859→          state['chg' + (key === 'btc' ? 'Btc' : key === 'eth' ? 'Eth' : '')] = d[apiKey].usd_24h_change;
   860→        });
   861→        render();
   862→        setLive(true, 'LIVE — polling every 45s');
   863→      })
   864→      .catch(function () {
   865→        clearTimeout(t);
   866→        feedOk = false;
   867→        if (Object.keys(state).length > 0) setLive(false, 'offline — last known price');
   868→        else setLive(false, 'offline — cannot reach price feed');
   869→      });
   870→  }
