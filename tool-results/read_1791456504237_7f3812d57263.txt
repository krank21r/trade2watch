    10→// ─── Trade Confirmed board ───────────────────────────────────────────────────
    11→// A dedicated page that only shows trades whose entry is confirmed. Two ways
    12→// to confirm: price is inside the entry zone right now (state 'LIVE'), or the
    13→// zone was HIT by a candle wick recently (state 'HIT') — an entry hit counts
    14→// as a confirmed trade even after price moves on. Shares the watchlist with
    15→// the Trade Setups tab via localStorage ('tw_watchlist') and keeps an
    16→// append-only confirmation log ('tw_confirmed_log') so entries confirmed
    17→// earlier stay visible after price moves on.
    18→
    19→const POLL_MS = 45_000
    20→const LOG_KEY = 'tw_confirmed_log'
    21→const MAX_LOG = 24
    22→
    23→interface ConfirmEvent {
    24→  key: string // "SYMBOL:side"
    25→  symbol: string
    26→  market: 'crypto' | 'stock'
    27→  displayName: string
    28→  pair: string
    29→  currency: string
    30→  side: 'long' | 'short'
    31→  tag: string
    32→  strategy: string
    33→  entryLow: number
    34→  entryHigh: number
    35→  stop: number
    36→  t1: number
    37→  t2: number
    38→  runner: number | null
    39→  rr: string
    40→  confirmedAt: number
    41→  confirmedPrice: number
    42→  lastSeenAt: number
    43→  lastPrice: number
    44→  exitedAt: number | null // set when price left the entry zone
    45→  via?: 'spot' | 'wick' // wick = discovered from candle history (spot polls missed it)
    46→  loggedAt?: number // when this row was created (dedupe window for wick hits)
    47→}
    48→
    49→function lsGet(k: string, d: string): string {
    50→  try {
    51→    return localStorage.getItem(k) ?? d
    52→  } catch {
    53→    return d
    54→  }
    55→}
    56→
    57→function loadLog(): ConfirmEvent[] {
    58→  try {
    59→    const raw = JSON.parse(localStorage.getItem(LOG_KEY) ?? '[]') as ConfirmEvent[]
    60→    return Array.isArray(raw) ? raw.filter((e) => e && typeof e.key === 'string') : []
    61→  } catch {
    62→    return []
    63→  }
    64→}
    65→
    66→function saveLog(events: ConfirmEvent[]) {
    67→  try {
    68→    localStorage.setItem(LOG_KEY, JSON.stringify(events))
    69→  } catch {
    70→    /* private mode */
    71→  }
    72→}
    73→
    74→const fmtTime = (t: number) =>
    75→  new Date(t).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' })
    76→
    77→/** time-only for today, date+time otherwise — wick hits are often hours/days old */
    78→const fmtWhen = (t: number) => {
    79→  const d = new Date(t)
    80→  const now = new Date()
    81→  const sameDay =
    82→    d.getFullYear() === now.getFullYear() && d.getMonth() === now.getMonth() && d.getDate() === now.getDate()
    83→  return sameDay
    84→    ? d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
    85→    : d.toLocaleString([], { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
    86→}
    87→
    88→const fmtDur = (ms: number) => {
    89→  const m = Math.max(0, Math.round(ms / 60_000))
    90→  return m < 60 ? `${m}m` : `${Math.floor(m / 60)}h ${m % 60}m`
    91→}
    92→
    93→function tagCls(tag: string): string {
    94→  if (tag === 'PREFERRED') return 'border-warn/40 bg-warn/10 text-warn'
    95→  if (tag === 'COUNTER-TREND') return 'border-info/35 bg-info/10 text-info'
    96→  return 'border-tv-line bg-tv-panel2 text-tv-muted'
    97→}

    99→// ─── plan ladder row ─────────────────────────────────────────────────────────

   101→function LadderRow({
   102→  icon, label, sub, value, cls, accent,
   103→}: {
   104→  icon: string
   105→  label: string
   106→  sub?: string
   107→  value: string
   108→  cls: string
   109→  accent?: boolean
   110→}) {
   111→  const rowCls = accent
   112→    ? 'border-warn/40 bg-warn/5'
   113→    : cls === 'text-bear'
   114→      ? 'border-bear/30 bg-bear/5'
   115→      : 'border-tv-line bg-tv-panel2/50'
   116→  const soft = accent ? 'var(--warn-soft)' : cls === 'text-bear' ? 'var(--bear-soft)' : 'var(--info-soft)'
   117→  return (
   118→    <div role="listitem" className={`flex items-center gap-3 rounded-xl border px-3 py-2 ${rowCls}`}>
   119→      <span
   120→        className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-[13px]"
   121→        style={{ background: soft }}
   122→        aria-hidden="true"
   123→      >
   124→        {icon}
   125→      </span>
   126→      <div className="min-w-0 flex-1">
   127→        <div className="text-[10.5px] font-bold uppercase tracking-wider text-tv-muted">{label}</div>
   128→        {sub && <div className="mt-0.5 truncate text-[10px] text-tv-muted2">{sub}</div>}
   129→      </div>
   130→      <div className={`shrink-0 text-[14.5px] font-bold tabular-nums ${cls}`}>{value}</div>
   131→    </div>
   132→  )
   133→}

   137→function ConfirmedCard({
   138→  setup, s, kind, event, sizingTxt, index,
   139→}: {
   140→  setup: BoardSetup
   141→  s: SideSetup
   142→  kind: 'long' | 'short'
   143→  event: ConfirmEvent | undefined
   144→  sizingTxt: string | null
   145→  index: number
   146→}) {
   147→  const isLong = kind === 'long'
   148→  const accent = isLong ? 'text-bull' : 'text-bear'
   149→  const [copied, setCopied] = useState(false)
   150→
   151→  const copyPlan = async () => {
   152→    const runner = s.runner ? ` · Runner ${fmtPrice(s.runner, setup.currency)}` : ''
   153→    const txt =
   154→      (s.state === 'HIT'
   155→        ? `${setup.pair} ${kind.toUpperCase()} trade confirmed — ENTRY HIT by wick ${fmtPrice(
   156→            s.touchedPrice,
   157→            setup.currency,
   158→          )}${s.touchedAt !== null ? ` ${fmtWhen(s.touchedAt)}` : ''}`
   159→        : `${setup.pair} ${kind.toUpperCase()} trade confirmed` + (event ? ` at ${fmtTime(event.confirmedAt)}` : '')) +
   160→      ` — entry ${fmtPrice(s.entryLow, setup.currency)} to ${fmtPrice(s.entryHigh, setup.currency)}` +
   161→      ` · SL ${fmtPrice(s.stop, setup.currency)}` +
   162→      ` · TP1 ${fmtPrice(s.t1, setup.currency)} · TP2 ${fmtPrice(s.t2, setup.currency)}${runner} · RR ${s.rr}`
   163→    try {
   164→      await navigator.clipboard.writeText(txt)
   165→      setCopied(true)
   166→      setTimeout(() => setCopied(false), 1600)
   167→    } catch {
   168→      /* clipboard unavailable */
   169→    }
   170→  }
   171→
   172→  return (
   173→    <motion.article
   174→      initial={{ opacity: 0, y: 8 }}
   175→      animate={{ opacity: 1, y: 0 }}
   176→      transition={{ duration: 0.25, ease: 'easeOut', delay: Math.min(index, 8) * 0.05 }}
   177→      className={`relative overflow-hidden rounded-3xl border p-4 sm:p-5 ${isLong ? 'border-bull/25' : 'border-bear/25'}`}
   178→      style={{
   179→        background: `linear-gradient(180deg, var(--tv-panel) 55%, ${isLong ? 'var(--bull-soft)' : 'var(--bear-soft)'})`,
   180→        boxShadow: isLong ? 'var(--card-glow)' : 'var(--card-glow-bear)',
   181→      }}
   182→    >
   183→      {/* featured top hairline — teal → teal-mid gradient (rose for shorts) */}
   184→      <span
   185→        aria-hidden
   186→        className="absolute inset-x-0 top-0 h-px"
   187→        style={{
   188→          background: isLong
   189→            ? 'linear-gradient(90deg, transparent, var(--bull), var(--info), var(--bull), transparent)'
   190→            : 'linear-gradient(90deg, transparent, var(--bear), var(--bear-soft), var(--bear), transparent)',
   191→        }}
   192→      />
   193→      {/* header */}
   194→      <div className="flex flex-wrap items-start justify-between gap-3">
   195→        <div className="flex items-center gap-3">
   196→          <span className="relative flex h-9 w-9 shrink-0 items-center justify-center" aria-hidden="true">
   197→            <span className={`absolute inline-flex h-full w-full animate-ping rounded-full ${isLong ? 'bg-bull/30' : 'bg-bear/30'}`} />
   198→            <span
   199→              className={`relative inline-flex h-9 w-9 items-center justify-center rounded-full text-[15px] font-bold ${isLong ? 'bg-bull/20 text-bull' : 'bg-bear/20 text-bear'}`}
   200→            >
   201→              ✓
   202→            </span>
   203→          </span>
   204→          <div>
   205→            <div className="flex flex-wrap items-center gap-2">
   206→              <h3 className="text-[15.5px] font-bold tracking-tight text-tv-ink">{setup.pair}</h3>
   207→              <span className={`rounded-full border px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide ${isLong ? 'border-bull/40 bg-bull/10 text-bull' : 'border-bear/40 bg-bear/10 text-bear'}`}>
   208→                {kind} confirmed
   209→              </span>
   210→              <span className={`rounded-full border px-2 py-0.5 text-[10px] font-bold ${tagCls(s.tag)}`}>{s.tag}</span>
   211→            </div>
   212→            <div className="mt-0.5 text-[11.5px] text-tv-muted">
   213→              {setup.displayName} · {setup.regime.toLowerCase().replace('_', ' ')} · {setup.timeframe}
   214→            </div>
   215→          </div>
   216→        </div>
   217→        <div className="text-right">
   218→          <div className="text-[19px] font-bold tabular-nums tracking-tight text-tv-ink">
   219→            {fmtPrice(setup.price, setup.currency)}
   220→          </div>
   221→          <span
   222→            className={`mt-0.5 inline-block rounded-full px-2 py-0.5 text-[10.5px] font-bold tabular-nums ${
   223→              setup.change24h >= 0 ? 'bg-bull/10 text-bull' : 'bg-bear/10 text-bear'
   224→            }`}
   225→          >
   226→            {setup.change24h >= 0 ? '+' : ''}
   227→            {setup.change24h.toFixed(2)}% 24h
   228→          </span>
   229→        </div>
   230→      </div>
   231→
   232→      {/* live line */}
   233→      <div className="mt-3 flex flex-wrap items-center gap-2 text-[11.5px]">
   234→        {s.state === 'HIT' ? (
   235→          <span className="inline-flex items-center gap-1.5 rounded-full border border-warn/40 bg-warn/10 px-2.5 py-1 font-bold text-warn">
   236→            <span className="inline-block h-1.5 w-1.5 rounded-full bg-warn" aria-hidden="true" />
   237→            🎯 ENTRY HIT
   238→          </span>
   239→        ) : (
   240→          <span className={`inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 font-bold ${isLong ? 'border-bull/40 bg-bull/10 text-bull' : 'border-bear/40 bg-bear/10 text-bear'}`}>
   241→            <span className={`inline-block h-1.5 w-1.5 animate-pulse rounded-full ${isLong ? 'bg-bull' : 'bg-bear'}`} aria-hidden="true" />
   242→            IN ZONE
   243→          </span>
   244→        )}
   245→        <span className="text-tv-muted">
   246→          {s.state === 'HIT' ? (
   247→            <>
   248→              wick {fmtPrice(s.touchedPrice, setup.currency)} hit {s.touchedAt !== null ? fmtWhen(s.touchedAt) : '—'} · price now{' '}
   249→              {fmtPrice(setup.price, setup.currency)} ({s.distPct?.toFixed(1) ?? '?'}% away) — entry triggered
   250→            </>
   251→          ) : event ? (
   252→            <>
   253→              confirmed {fmtTime(event.confirmedAt)} · in zone {fmtDur(Date.now() - event.confirmedAt)}
   254→            </>
   255→          ) : (
   256→            'price inside entry zone'
   257→          )}
   258→        </span>
   259→      </div>
   260→
   261→      {/* ladder */}
   262→      <div role="list" aria-label={`${setup.pair} ${kind} plan`} className="mt-3 space-y-1.5">
   263→        <LadderRow
   264→          icon="◆"
   265→          label="Entry zone"
   266→          sub={s.strategy}
   267→          value={`${fmtPrice(s.entryLow, setup.currency)} – ${fmtPrice(s.entryHigh, setup.currency)}`}
   268→          cls="text-warn"
   269→          accent
   270→        />
   271→        <LadderRow icon="✕" label="Stop loss" sub={s.stopNote} value={fmtPrice(s.stop, setup.currency)} cls="text-bear" />
   272→        <LadderRow icon="◎" label="Target 1" value={fmtPrice(s.t1, setup.currency)} cls="text-info" />
   273→        <LadderRow icon="◎" label="Target 2" value={fmtPrice(s.t2, setup.currency)} cls="text-info" />
   274→        {s.runner !== null && (
   275→          <LadderRow icon="→" label="Runner" value={fmtPrice(s.runner, setup.currency)} cls="text-info" />
   276→        )}
   277→      </div>
   278→
   279→      {/* footer */}
   280→      <div className="mt-3 flex flex-wrap items-center justify-between gap-2 border-t border-tv-div pt-3">
   281→        <div className="min-w-0 text-[11px] text-tv-muted">
   282→          <span className="rounded-full border border-tv-line bg-tv-panel2 px-2 py-0.5 font-bold tabular-nums text-tv-ink">
   283→            R:R {s.rr}
   284→          </span>
   285→          {sizingTxt && <span className="ml-2 tabular-nums">{sizingTxt}</span>}
   286→        </div>
   287→        <button
   288→          onClick={copyPlan}
   289→          suppressHydrationWarning
   290→          aria-label={`Copy ${setup.pair} ${kind} trade plan`}
   291→          className="shrink-0 rounded-xl border border-tv-line bg-tv-panel2 px-3 py-2 text-[12px] font-semibold text-tv-ink transition hover:border-tv-line-strong focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-bull/50"
   292→        >
   293→          {copied ? '✓ Copied' : '⧉ Copy plan'}
   294→        </button>
   295→      </div>
   296→    </motion.article>
   297→  )
   298→}

   300→// ─── board ───────────────────────────────────────────────────────────────────

   302→export function ConfirmedBoard() {
   303→  const [symbols, setSymbols] = useState<string[]>(DEFAULT_WATCHLIST)
   304→  const [data, setData] = useState<SetupsPayload | null>(null)
   305→  const [loading, setLoading] = useState(true)
   306→  const [error, setError] = useState<string | null>(null)
   307→  const [log, setLog] = useState<ConfirmEvent[]>([])
   308→  const [acct, setAcct] = useState('')
   309→  const [riskPct, setRiskPct] = useState('1')
   310→  const mounted = useRef(true)
   311→  const logRef = useRef<ConfirmEvent[]>([])
   312→  const wlLoaded = useRef(false)
   313→  const { toast } = useToast()
   314→
   315→  useEffect(() => {
   316→    mounted.current = true
   317→    setAcct(lsGet('tw_acct', ''))
   318→    setRiskPct(lsGet('tw_risk', '1'))
   319→    const wl = loadWatchlist()
   320→    setSymbols(wl)
   321→    saveWatchlist(wl)
   322→    wlLoaded.current = true
   323→    const l = loadLog()
   324→    logRef.current = l
   325→    setLog(l)
   326→    return () => {
   327→      mounted.current = false
   328→    }
   329→  }, [])
   330→
   331→  // share watchlist edits with the Trade Setups tab (persist list)
   332→  useEffect(() => {
   333→    if (!wlLoaded.current) return
   334→    saveWatchlist(symbols)
   335→  }, [symbols])
   336→
   337→  const load = useCallback(async () => {
   338→    try {
   339→      const res = await fetch(`/api/setups?symbols=${symbols.join(',')}`, { cache: 'no-store' })
   340→      if (!res.ok) throw new Error(`API ${res.status}`)
   341→      const d = (await res.json()) as SetupsPayload
   342→      if (!mounted.current) return
   343→      setData(d)
   344→      setError(null)
   345→    } catch (e) {
   346→      if (mounted.current) setError(String(e).slice(0, 120))
   347→    } finally {
   348→      if (mounted.current) setLoading(false)
   349→    }
   350→  }, [symbols])
   351→
   352→  useEffect(() => {
   353→    setLoading(true)
   354→    load()
   355→    const t = setInterval(load, POLL_MS)
   356→    return () => clearInterval(t)
   357→  }, [load])
   358→
   359→  // confirmation tracker — append new LIVE entries to the log, mark exits.
   360→  // Also reconciles WICK HITS the spot polls missed: the server checks real
   361→  // candle highs/lows against each entry zone, so a touch that happened
   362→  // between polls (or while the tab was closed) still gets logged here.
   363→  useEffect(() => {
   364→    if (!data) return
   365→    const now = Date.now()
   366→    const next = logRef.current.map((e) => ({ ...e }))
   367→    const fired: ConfirmEvent[] = []
   368→    const liveKeys = new Set<string>()
   369→    for (const setup of data.setups) {
   370→      for (const kind of ['long', 'short'] as const) {
   371→        const s = setup[kind]
   372→        if (!s) continue
   373→        const key = `${setup.symbol}:${kind}`
   374→        const isLive = s.state === 'LIVE'
   375→        if (isLive) {
   376→          liveKeys.add(key)
   377→          const existing = next.find((e) => e.key === key && !e.exitedAt)
   378→          if (existing) {
   379→            existing.lastSeenAt = now
   380→            existing.lastPrice = setup.price
   381→          } else {
   382→            const ev: ConfirmEvent = {
   383→              key,
   384→              symbol: setup.symbol,
   385→              market: setup.market,
   386→              displayName: setup.displayName,
   387→              pair: setup.pair,
   388→              currency: setup.currency,
   389→              side: kind,
   390→              tag: s.tag,
   391→              strategy: s.strategy,
   392→              entryLow: s.entryLow,
   393→              entryHigh: s.entryHigh,
   394→              stop: s.stop,
   395→              t1: s.t1,
   396→              t2: s.t2,
   397→              runner: s.runner,
   398→              rr: s.rr,
   399→              confirmedAt: now,
   400→              confirmedPrice: setup.price,
   401→              lastSeenAt: now,
   402→              lastPrice: setup.price,
   403→              exitedAt: null,
   404→              via: 'spot',
   405→              loggedAt: now,
   406→            }
   407→            next.unshift(ev)
   408→            fired.push(ev)
   409→          }
   410→        } else if (s.touchedAt !== null && s.touchedPrice !== null) {
   411→          // price is NOT in the zone now, but a recent candle wick was —
   412→          // record it as history (already exited). Dedupe: same candle touch
   413→          // yields the same touchedAt; also suppress re-logs within 6h of the
   414→          // last recorded wick for this key so drifting zones can't spam rows.
   415→          const dupe = next.some(
   416→            (e) =>
   417→              e.key === key &&
   418→              (e.confirmedAt === s.touchedAt ||
   419→                (e.via === 'wick' && typeof e.loggedAt === 'number' && now - e.loggedAt < 6 * 3_600_000)),
   420→          )
   421→          if (!dupe) {
   422→            const ev: ConfirmEvent = {
   423→              key,
   424→              symbol: setup.symbol,
   425→              market: setup.market,
   426→              displayName: setup.displayName,
   427→              pair: setup.pair,
   428→              currency: setup.currency,
   429→              side: kind,
   430→              tag: s.tag,
   431→              strategy: s.strategy,
   432→              entryLow: s.entryLow,
   433→              entryHigh: s.entryHigh,
   434→              stop: s.stop,
   435→              t1: s.t1,
   436→              t2: s.t2,
   437→              runner: s.runner,
   438→              rr: s.rr,
   439→              confirmedAt: s.touchedAt,
   440→              confirmedPrice: s.touchedPrice,
   441→              lastSeenAt: s.touchedAt,
   442→              lastPrice: s.touchedPrice,
   443→              exitedAt: now,
   444→              via: 'wick',
   445→              loggedAt: now,
   446→            }
   447→            next.unshift(ev)
   448→            fired.push(ev) // toast below — the user deserves to know the zone was hit
   449→          }
   450→        }
   451→      }
   452→    }
   453→    let changed = fired.length > 0
   454→    for (const e of next) {
   455→      if (!e.exitedAt && !liveKeys.has(e.key)) {
   456→        e.exitedAt = now
   457→        changed = true
   458→      }
   459→    }
   460→    if (!changed) return
   461→    const trimmed = next.slice(0, MAX_LOG)
   462→    logRef.current = trimmed
   463→    setLog(trimmed)
   464→    saveLog(trimmed)
   465→    for (const f of fired) {
   466→      if (f.via === 'wick') {
   467→        toast({
   468→          title: `🎯 ${f.symbol} ${f.side.toUpperCase()} zone was hit`,
   469→          description: `A candle wick reached ${fmtPrice(f.confirmedPrice, f.currency)} inside the entry zone ${fmtPrice(
   470→            f.entryLow,
   471→            f.currency,
   472→          )} – ${fmtPrice(f.entryHigh, f.currency)} (${fmtWhen(f.confirmedAt)}) — spot polls missed it`,
   473→        })
   474→      } else {
   475→        toast({
   476→          title: `✅ ${f.symbol} ${f.side.toUpperCase()} CONFIRMED`,
   477→          description: `Price ${fmtPrice(f.confirmedPrice, f.currency)} is inside the entry zone ${fmtPrice(
   478→            f.entryLow,
   479→            f.currency,
   480→          )} – ${fmtPrice(f.entryHigh, f.currency)} · SL ${fmtPrice(f.stop, f.currency)}`,
   481→        })
   482→      }
   483→    }
   484→  }, [data, toast])
   485→
   486→  // ── derive confirmed trades ──
   487→  // LIVE = price inside the zone right now · HIT = entry triggered by a wick,
   488→  // price has since moved on. Both are confirmed trades and render as cards.
   489→  type CardItem = {
   490→    key: string
   491→    setup: BoardSetup
   492→    s: SideSetup
   493→    kind: 'long' | 'short'
   494→    event: ConfirmEvent | undefined
   495→    sizingTxt: string | null
   496→  }
   497→  const aNum = parseFloat(acct)
   498→  const rNum = parseFloat(riskPct)
   499→  const sizingFor = (setup: BoardSetup, s: SideSetup): string | null => {
   500→    if (!(aNum > 0) || !(rNum > 0)) return null
   501→    const m = (s.entryLow + s.entryHigh) / 2
   502→    const dist = Math.abs(m - s.stop)
   503→    if (!(dist > 0)) return null
   504→    const units = (aNum * rNum) / 100 / dist
   505→    const unitName = setup.market === 'crypto' ? setup.symbol : 'shares'
   506→    return `size ≈ ${
   507→      units >= 100 ? Math.round(units).toLocaleString('en-US') : Math.round(units * 10000) / 10000
   508→    } ${unitName} · risk $${Math.round((aNum * rNum) / 100).toLocaleString('en-US')}`
   509→  }
   510→  const live: CardItem[] = []
   511→  const hits: CardItem[] = []
   512→  for (const setup of data?.setups ?? []) {
   513→    for (const kind of ['long', 'short'] as const) {
   514→      const s = setup[kind]
   515→      if (!s) continue
   516→      if (s.state === 'LIVE') {
   517→        const event = log.find((e) => e.key === `${setup.symbol}:${kind}` && !e.exitedAt)
   518→        live.push({ key: `${setup.symbol}:${kind}`, setup, s, kind, event, sizingTxt: sizingFor(setup, s) })
   519→      } else if (s.state === 'HIT') {
   520→        hits.push({ key: `${setup.symbol}:${kind}`, setup, s, kind, event: undefined, sizingTxt: sizingFor(setup, s) })
   521→      }
   522→    }
   523→  }
   524→  const confirmedCount = live.length + hits.length
   525→  // hide history rows that duplicate a currently displayed HIT card
   526→  const hitTouchIds = new Set(hits.map((h) => `${h.key}:${h.s.touchedAt}`))
   527→  const history = log.filter((e) => e.exitedAt && !hitTouchIds.has(`${e.key}:${e.confirmedAt}`))
   528→  const now = Date.now()
   529→
   530→  return (
   531→    <div className="min-h-full mx-auto flex w-full max-w-[1400px] flex-col gap-5 px-4 py-5 sm:px-6 sm:py-6">
   532→      {/* header */}
   533→      <div className="flex flex-wrap items-center gap-2 rounded-2xl border border-tv-line bg-tv-panel shadow-sm p-3 sm:p-4">
   534→        <div className="mr-1 flex items-center gap-2">
   535→          <span
   536→            className="inline-flex h-8 w-8 items-center justify-center rounded-xl border text-[14px]"
   537→            style={{ background: 'var(--bull-soft)', borderColor: 'var(--bull-line)' }}
   538→            aria-hidden="true"
   539→          >
   540→            ✅
   541→          </span>
   542→          <div className="text-[11px] font-bold uppercase tracking-[0.14em] text-tv-muted">
   543→            Trade confirmed
   544→          </div>
   545→        </div>
   546→        <span
   547→          role="status"
   548→          aria-live="polite"
   549→          className={`rounded-full border px-2.5 py-1 text-[11px] font-bold tabular-nums ${
   550→            confirmedCount ? 'border-bull/40 bg-bull/10 text-bull' : 'border-tv-line bg-tv-panel2 text-tv-muted'
   551→          }`}
   552→        >
   553→          {confirmedCount
   554→            ? `${confirmedCount} confirmed now${live.length && hits.length ? ` · ${live.length} in zone · ${hits.length} hit` : ''}`
   555→            : '0 confirmed now'}
   556→        </span>
   557→        <div className="flex flex-wrap items-center gap-1.5">
   558→          {symbols.map((sym) => (
   559→            <span
   560→              key={sym}
   561→              className="rounded-full border border-tv-line bg-tv-panel2 px-2.5 py-1 text-[11px] font-semibold text-tv-muted"
   562→            >
   563→              {sym}
   564→            </span>
   565→          ))}
   566→        </div>
   567→        <div className="ml-auto text-[11px] tabular-nums text-tv-muted">
   568→          {loading ? 'refreshing…' : `refreshed ${data ? fmtTime(new Date(data.updatedAt).getTime()) : '—'} · auto every 45s`}
   569→        </div>
   570→      </div>
   571→
   572→      {error && (
   573→        <div role="alert" className="rounded-xl border px-3 py-2.5 text-[13px]" style={{ background: 'var(--bear-soft)', borderColor: 'var(--bear-line)' }}>
   574→          <span className="text-bear">{error}</span>
   575→        </div>
   576→      )}
   577→
   578→      {/* confirmed trades — in zone + entry hits */}
   579→      <section aria-label="Confirmed entries" className="flex flex-col gap-4">
   580→        {loading && !data ? (
   581→          <>
   582→            <div className="h-[280px] animate-pulse rounded-2xl border border-tv-line bg-tv-panel" />
   583→            <div className="h-[280px] animate-pulse rounded-2xl border border-tv-line bg-tv-panel" />
   584→          </>
   585→        ) : confirmedCount === 0 ? (
   586→          <div className="rounded-2xl border border-dashed border-tv-line-strong bg-tv-panel2/50 p-10 text-center">
   587→            <div className="text-[15px] font-semibold text-tv-ink">No confirmed entries right now</div>
   588→            <div className="mx-auto mt-1.5 max-w-md text-[12.5px] leading-relaxed text-tv-muted">
   589→              The tracker is watching {symbols.length} symbol{symbols.length === 1 ? '' : 's'} (
   590→              {symbols.join(', ')}). The moment price touches an entry zone — or a candle wick hits one —
   591→              the trade is confirmed and appears here automatically.
   592→            </div>
   593→          </div>
   594→        ) : (
   595→          <>
   596→            {live.length > 0 && (
   597→              <div className="flex flex-col gap-3">
   598→                <div className="text-[11px] font-bold uppercase tracking-[0.14em] text-tv-muted">
   599→                  In entry zone right now <span className="text-warn">— price inside the zone</span>
   600→                </div>
   601→                <div className="grid gap-4 lg:grid-cols-2">
   602→                  {live.map((c, i) => (
   603→                    <ConfirmedCard key={c.key} setup={c.setup} s={c.s} kind={c.kind} event={c.event} sizingTxt={c.sizingTxt} index={i} />
   604→                  ))}
   605→                </div>
   606→              </div>
   607→            )}
   608→            {hits.length > 0 && (
   609→              <div className="flex flex-col gap-3">
   610→                <div className="text-[11px] font-bold uppercase tracking-[0.14em] text-tv-muted">
   611→                  Entry hit <span className="text-warn">— zone triggered by a candle wick, price moved on · still confirmed</span>
   612→                </div>
   613→                <div className="grid gap-4 lg:grid-cols-2">
   614→                  {hits.map((c, i) => (
   615→                    <ConfirmedCard key={c.key} setup={c.setup} s={c.s} kind={c.kind} event={c.event} sizingTxt={c.sizingTxt} index={i} />
   616→                  ))}
   617→                </div>
   618→              </div>
   619→            )}
   620→          </>
   621→        )}
   622→      </section>
   623→
   624→      {/* history */}
   625→      {history.length > 0 && (
   626→        <section aria-label="Earlier confirmations" className="flex flex-col gap-3">
   627→          <div className="flex items-center justify-between gap-2">
   628→            <div className="text-[11px] font-bold uppercase tracking-[0.14em] text-tv-muted">
   629→              Earlier <span className="text-tv-muted2">— confirmed, then price moved on</span>
   630→            </div>
   631→            <button
   632→              onClick={() => {
   633→                logRef.current = []
   634→                setLog([])
   635→                saveLog([])
   636→              }}
   637→              suppressHydrationWarning
   638→              aria-label="Clear confirmation history"
   639→              className="rounded-xl border border-tv-line bg-tv-panel2 px-3 py-1.5 text-[11.5px] font-semibold text-tv-muted transition hover:text-tv-ink hover:border-tv-line-strong focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-bull/50"
   640→            >
   641→              Clear history
   642→            </button>
   643→          </div>
   644→          <div className="flex flex-col gap-2">
   645→            {history.slice(0, 8).map((e) => (
   646→              <div
   647→                key={`${e.key}:${e.confirmedAt}`}
   648→                className="flex flex-wrap items-center gap-2.5 rounded-xl border border-tv-line bg-tv-panel px-3 py-2.5 text-[12px]"
   649→              >
   650→                <span className="font-bold text-tv-ink">{e.pair}</span>
   651→                <span
   652→                  className={`rounded-full border px-2 py-0.5 text-[10px] font-bold uppercase ${
   653→                    e.side === 'long' ? 'border-bull/35 bg-bull/10 text-bull' : 'border-bear/35 bg-bear/10 text-bear'
   654→                  }`}
   655→                >
   656→                  {e.side}
   657→                </span>
   658→                {e.via === 'wick' ? (
   659→                  <span className="tabular-nums text-tv-muted">
   660→                    🎯 wick {fmtPrice(e.confirmedPrice, e.currency)} hit {fmtWhen(e.confirmedAt)} · zone {fmtPrice(
   661→                      e.entryLow,
   662→                      e.currency,
   663→                    )} – {fmtPrice(e.entryHigh, e.currency)} · last {fmtPrice(e.lastPrice, e.currency)}
   664→                  </span>
   665→                ) : (
   666→                  <span className="tabular-nums text-tv-muted">
   667→                    confirmed {fmtTime(e.confirmedAt)} · in zone {fmtDur((e.exitedAt ?? e.lastSeenAt) - e.confirmedAt)} ·
   668→                    exited {fmtTime(e.exitedAt ?? now)} · last {fmtPrice(e.lastPrice, e.currency)}
   669→                  </span>
   670→                )}
   671→                <span className="ml-auto rounded-full border border-tv-line bg-tv-panel2 px-2 py-0.5 text-[10px] font-bold text-tv-muted">
   672→                  {e.tag}
   673→                </span>
   674→              </div>
   675→            ))}
   676→          </div>
   677→        </section>
   678→      )}
   679→
   680→      {/* sizing hint */}
   681→      {confirmedCount > 0 && !(aNum > 0 && rNum > 0) && (
   682→        <div className="text-[11.5px] text-tv-muted">
   683→          Tip: set your account size and risk % on the Trade Setups page to see suggested position sizes here.
   684→        </div>
   685→      )}
   686→    </div>
   687→  )
   688→}
