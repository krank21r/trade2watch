    75→/** One-chip summary per asset for the ZoneWatch strip. */
    76→function summarize(s: BoardSetup): { cls: string; txt: string } {
    77→  const l = s.long
    78→  const sh = s.short
    79→  if (sh?.state === 'LIVE') return { cls: chipCls.liveShort, txt: 'SHORT CONFIRMED' }
    80→  if (l?.state === 'LIVE') return { cls: chipCls.liveLong, txt: 'LONG CONFIRMED' }
    81→  // entry hit = confirmed: zone was triggered by a wick, price moved on
    82→  if (sh?.state === 'HIT') return { cls: chipCls.hitShort, txt: 'SHORT HIT — CONFIRMED' }
    83→  if (l?.state === 'HIT') return { cls: chipCls.hitLong, txt: 'LONG HIT — CONFIRMED' }
    84→  const waits: string[] = []
    85→  if (l?.state === 'WAITING' && l.distPct !== null) waits.push(`${l.distPct.toFixed(1)}% to LONG`)
    86→  if (sh?.state === 'WAITING' && sh.distPct !== null) waits.push(`${sh.distPct.toFixed(1)}% to SHORT`)
    87→  const voids: string[] = []
    88→  if (l?.state === 'VOID') voids.push('LONG void')
    89→  if (sh?.state === 'VOID') voids.push('SHORT void')
    90→  if (voids.length === 2) return { cls: chipCls.void, txt: 'SETUP VOID' }
    91→  if (waits.length === 0 && voids.length === 1) return { cls: chipCls.void, txt: `${voids[0].toUpperCase()} — WAIT` }
    92→  if (voids.length === 1) return { cls: chipCls.waiting, txt: `${waits[0]} · ${voids[0]}` }
    93→  return { cls: chipCls.waiting, txt: waits.length ? `WAITING · ${waits.join(' · ')}` : 'WAITING' }
    94→}

    98→function pctFrom(mid: number, v: number): string {
    99→  const p = ((v - mid) / mid) * 100
   100→  return `${p >= 0 ? '+' : ''}${p.toFixed(1)}%`
   101→}
   102→
   103→function rMult(v: number, mid: number, risk: number): string {
   104→  return risk > 0 ? `${(Math.abs(v - mid) / risk).toFixed(1)}R` : '—'
   105→}

   107→/** time-only for today, date+time otherwise — wick hits are often hours/days old */
   108→function fmtWhen(t: number): string {
   109→  const d = new Date(t)
   110→  const now = new Date()
   111→  const sameDay =
   112→    d.getFullYear() === now.getFullYear() && d.getMonth() === now.getMonth() && d.getDate() === now.getDate()
   113→  return sameDay
   114→    ? d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
   115→    : d.toLocaleString([], { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
   116→}
